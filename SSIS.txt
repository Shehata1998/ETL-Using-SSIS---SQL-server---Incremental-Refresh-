        // ─────────────────────────────────────────────────────────────────────
        // Runs a single Python script and returns true = success, false = failed
        // ─────────────────────────────────────────────────────────────────────
        private bool RunScript(string scriptPath, string logFile,
                               string label, bool fireAgain)
        {
            string stdOut = string.Empty;
            string stdErr = string.Empty;
            int exitCode = -1;

            try
            {
                // Validate script file exists
                if (!File.Exists(scriptPath))
                {
                    Dts.Events.FireError(2, label,
                        "Script not found: " + scriptPath, "", 0);
                    return false;
                }

                // Delete old log before run
                if (File.Exists(logFile))
                    File.Delete(logFile);

                Dts.Events.FireInformation(0, label,
                    "Starting: " + scriptPath, "", 0, ref fireAgain);

                // Launch Python
                ProcessStartInfo psi = new ProcessStartInfo();
                psi.FileName = PythonExe;
                psi.Arguments = "\"" + scriptPath + "\"";
                psi.RedirectStandardOutput = true;
                psi.RedirectStandardError = true;
                psi.UseShellExecute = false;
                psi.CreateNoWindow = true;
                psi.WorkingDirectory = WorkingDir;
                psi.EnvironmentVariables["PYTHONIOENCODING"] = "utf-8";
                psi.EnvironmentVariables["PYTHONUNBUFFERED"] = "1";

                using (Process process = new Process())
                {
                    process.StartInfo = psi;
                    process.Start();

                    stdOut = process.StandardOutput.ReadToEnd();
                    stdErr = process.StandardError.ReadToEnd();

                    bool finished = process.WaitForExit(TimeoutMs);
                    if (!finished)
                    {
                        process.Kill();
                        throw new Exception(label + " timed out after "
                            + (TimeoutMs / 1000) + "s.");
                    }
                    exitCode = process.ExitCode;
                }

                // Log stdout / stderr
                if (!string.IsNullOrWhiteSpace(stdOut))
                    Dts.Events.FireInformation(4, label + " stdout",
                        stdOut, "", 0, ref fireAgain);

                if (!string.IsNullOrWhiteSpace(stdErr))
                    Dts.Events.FireWarning(5, label + " stderr", stdErr, "", 0);

                // Read log file written by Python
                string logContent = ReadLogFile(logFile);
                Dts.Events.FireInformation(6, label + " log",
                    "ExitCode=" + exitCode + " | Log=" + logContent,
                    "", 0, ref fireAgain);

                // Check log content first (most reliable)
                if (logContent.Contains("SUCCESS"))
                    return true;

                if (logContent.Contains("FAILED"))
                {
                    Dts.Events.FireError(8, label,
                        "Script reported FAILED. Log: " + logContent, "", 0);
                    return false;
                }

                // Fall back to exit code
                if (exitCode != 0)
                {
                    Dts.Events.FireError(7, label,
                        "Exit code: " + exitCode + " | stdout: "
                        + stdOut.Substring(0, Math.Min(stdOut.Length, 1000)),
                        "", 0);
                    return false;
                }

                return true;
            }
            catch (Exception ex)
            {
                Dts.Events.FireError(99, label,
                    "Exception: " + ex.Message, "", 0);
                return false;
            }
        }

        // ─────────────────────────────────────────────────────────────────────
        private static string ReadLogFile(string path)
        {
            try
            {
                if (File.Exists(path))
                    return File.ReadAllText(path).Trim();
                return "(log file not found)";
            }
            catch (Exception ex)
            {
                return "(log read error: " + ex.Message + ")";
            }
        }
